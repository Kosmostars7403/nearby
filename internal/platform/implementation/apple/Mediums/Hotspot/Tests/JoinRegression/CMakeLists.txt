cmake_minimum_required(VERSION 3.16)
project(NearbyHotspotCancellationRegression LANGUAGES CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
if(NOT IS_DIRECTORY "${ABSEIL_SOURCE_DIR}" OR NOT IS_DIRECTORY "${GOOGLETEST_SOURCE_DIR}")
  message(FATAL_ERROR "Provide ABSEIL_SOURCE_DIR and GOOGLETEST_SOURCE_DIR; no dependencies are downloaded.")
endif()
get_filename_component(NEARBY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../../../.." ABSOLUTE)
add_subdirectory("${ABSEIL_SOURCE_DIR}" absl-build EXCLUDE_FROM_ALL)
add_subdirectory("${GOOGLETEST_SOURCE_DIR}" gtest-build EXCLUDE_FROM_ALL)
add_library(nearby_cancel "${NEARBY_ROOT}/internal/platform/cancellation_flag.cc")
target_include_directories(nearby_cancel PUBLIC "${NEARBY_ROOT}")
target_link_libraries(nearby_cancel PUBLIC absl::flat_hash_set absl::synchronization absl::time)
add_executable(scope_tests "${NEARBY_ROOT}/internal/platform/cancellation_scope_test.cc")
target_link_libraries(scope_tests PRIVATE nearby_cancel gtest_main)
enable_testing()
add_test(NAME cancellation_scope COMMAND scope_tests)
